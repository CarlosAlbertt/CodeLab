function mediaAprobados(notas: number[]): number {
const aprobadas = notas.filter((nota) => nota >= 5)
if (aprobadas.length === 0) return 0
const suma = aprobadas.reduce((total, nota) => total + nota, 0)
return suma / aprobadas.length
}
